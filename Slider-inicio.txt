EL PHP de WORDPRES
Creo que esun shortcode
/*=== GALERÍA ACF EN SWIPER PARA PÁGINA DE INICIO ===*/
function mostrar_galeria_inicio_shortcode($atts) {
    if (!function_exists('get_field')) {
        return 'El plugin ACF no está activo';
    }

    $galeria = get_field('galeria_inicio');
    if (empty($galeria)) {
        return 'No se encontró ninguna imagen en la galería.';
    }

    ob_start();

    // Incluir los CDN de Swiper
    echo '<link rel="stylesheet" href="https://cdn.jsdelivr.net/npm/swiper@11/swiper-bundle.min.css">';
    echo '<script src="https://cdn.jsdelivr.net/npm/swiper@11/swiper-bundle.min.js"></script>';

    echo '<div class="contenedor-galeria">';

    // Swiper principal con 3 imágenes por vista
    echo '<div class="swiper mySwiperInicio">';
    echo '<div class="swiper-wrapper">';

    foreach ($galeria as $imagen) {
        echo '<div class="swiper-slide">';
        echo '<a href="' . esc_url($imagen['url']) . '" class="popup-image">';
        echo '<img src="' . esc_url($imagen['url']) . '" alt="' . esc_attr($imagen['title']) . '" />';
        echo '</a>';
        echo '</div>';
    }

    echo '</div>'; // .swiper-wrapper
    echo '<div class="swiper-button-next"></div>';
    echo '<div class="swiper-button-prev"></div>';
    echo '</div>'; // .swiper
    echo '</div>'; // .contenedor-galeria

    // Script de inicialización de Swiper
    echo '<script>';
    echo 'document.addEventListener("DOMContentLoaded", function() {';
    echo 'var swiperInicio = new Swiper(".mySwiperInicio", {';
    echo 'slidesPerView: 3,';
    echo 'spaceBetween: 10,';
    echo 'loop: true,';
    echo 'autoplay: { delay: 3000, disableOnInteraction: false },';
    echo 'navigation: { nextEl: ".swiper-button-next", prevEl: ".swiper-button-prev" },';
    echo '});';
    echo '});';
    echo 'jQuery(document).ready(function($) {';
    echo '  $(".popup-image").magnificPopup({';
    echo '    type: "image",';
    echo '    gallery: { enabled: true }';
    echo '  });';
    echo '});';
    echo '</script>';

    return ob_get_clean();
}
add_shortcode('mostrar_galeria_inicio', 'mostrar_galeria_inicio_shortcode');
/*=== FIN GALERÍA ACF PARA PÁGINA DE INICIO ===*/

y se coloca con este shortcode: [mostrar_galeria_inicio]
