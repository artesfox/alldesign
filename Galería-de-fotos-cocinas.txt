/*=== GALERÍA ACF EN SWIPER PARA PRODUCTOS Y CPT ===*/
function mostrar_galeria_acf_swiper_shortcode($atts) {
    $atts = shortcode_atts(array(
        'producto_id' => '',
    ), $atts);
    
    $producto_id = $atts['producto_id'];
    
    if (empty($producto_id)) {
        global $post;
        $producto_id = $post->ID;
    }
    
    if (empty($producto_id)) {
        return 'No se pudo obtener el ID del producto o CPT.';
    }
    
    // Obtener imagen destacada
    $imagen_destacada_id = get_post_thumbnail_id($producto_id);
    $imagen_destacada = $imagen_destacada_id ? wp_get_attachment_image_src($imagen_destacada_id, 'full') : false;
    
    // Obtener galería ACF desde el campo "galeria_cocinas"
    $galeria = get_field('galeria_cocinas', $producto_id);
    
    ob_start();
    
    if ($imagen_destacada || $galeria) {
        // Incluir los CDN de Swiper
        echo '<link rel="stylesheet" href="https://cdn.jsdelivr.net/npm/swiper@11/swiper-bundle.min.css">';
        echo '<script src="https://cdn.jsdelivr.net/npm/swiper@11/swiper-bundle.min.js"></script>';

        echo '<div class="contenodie">';

        // Swiper principal
        echo '<div style="--swiper-navigation-color: #fff; --swiper-pagination-color: #fff" class="swiper mySwiper2">';
        echo '<div class="swiper-wrapper">';

  

        // Mostrar imágenes de la galería "galeria_cocinas"
        if ($galeria) {
            foreach ($galeria as $imagen) {
                echo '<div class="swiper-slide">';
                echo '<a href="' . esc_url($imagen['url']) . '" class="popup-image" style="background-image: url(' . esc_url($imagen['url']) . '); ">';
                //echo '<img src="' . esc_url($imagen['url']) . '" alt="' . esc_attr($imagen['alt']) . '" />';
                echo '</a>';
                echo '</div>';
            }
        }
        
        echo '</div>';
        echo '<div class="swiper-button-next"></div>';
        echo '<div class="swiper-button-prev"></div>';
        echo '</div>';

        // Swiper miniaturas
   // Verificar si hay más de una imagen en la galería antes de imprimir el contenedor de miniaturas
if ($galeria && count($galeria) > 1) {
    echo '<div thumbsSlider="" class="swiper mySwiper miniaturas-fx-galeria">';
    echo '<div class="swiper-wrapper">';

    // Miniaturas de la galería "galeria_cocinas"
    foreach ($galeria as $imagen) {
        echo '<div class="swiper-slide">';
        echo '<img src="' . esc_url($imagen['url']) . '" alt="' . esc_attr($imagen['alt']) . '" />';
        echo '</div>';
    }

    echo '</div>'; // Cierre de .swiper-wrapper
    echo '<div class="swiper-button-next mini-next"></div>';
    echo '<div class="swiper-button-prev mini-prev"></div>';
    echo '</div>'; // Cierre de .miniaturas-fx-galeria
}


        echo '</div>';

        // Script de inicialización de Swiper y Magnific Popup
        echo '<script>';
        echo 'document.addEventListener("DOMContentLoaded", function() {';
        echo 'var swiperThumbs = new Swiper(".mySwiper", {';
        echo 'spaceBetween: 10,';
        echo 'slidesPerView: 4,';
        echo 'freeMode: true,';
        echo 'watchSlidesProgress: true,';
        echo 'navigation: { nextEl: ".mini-next", prevEl: ".mini-prev" }';
        echo '});';
        echo 'var swiperMain = new Swiper(".mySwiper2", {';
        echo 'spaceBetween: 10,';
        echo 'navigation: { nextEl: ".swiper-button-next", prevEl: ".swiper-button-prev" },';
        echo 'thumbs: { swiper: swiperThumbs }';
        echo '});';
        echo '});';
        echo 'jQuery(document).ready(function($) {';
        echo '  $(".popup-image").magnificPopup({';
        echo '    type: "image",';
        echo '    gallery: { enabled: true }';
        echo '  });';
        echo '});';
        echo '</script>';
    } else {
        echo 'No se encontró ninguna imagen en la galería.';
    }

    return ob_get_clean();
}
add_shortcode('mostrar_galeria_acf_swiper', 'mostrar_galeria_acf_swiper_shortcode');
/*==== FIN GALERÍA ACF PARA PRODUCTOS ====*/
